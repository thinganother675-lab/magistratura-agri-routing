"""One bounded optional installed35B retry, then existing DEV60 if compatible."""
import asyncio
import json
from pathlib import Path
import sys
import time

ROOT=Path(__file__).resolve().parents[2]
sys.path.insert(0,str(ROOT/'src'));sys.path.insert(0,str(ROOT/'scripts'))
import study_c_live_20261004 as live
import mera_local_screen_20261003 as screen
import home_lm_api_benchmark as native
from agro_route.mera_screen_metrics_20261003 import summarize
from agro_route.home_config import resolve_api_key

MODEL='qwen3.5-35b-a3b'
SAFE=live.SAFE/'optional35'
RAW=live.RAW/'optional35'


async def main():
    SAFE.mkdir(exist_ok=False);RAW.mkdir(exist_ok=False)
    oldplan=json.loads((ROOT/'results/20261003_local_screen_571682f5/protocol_frozen.json').read_text(encoding='utf-8'))
    items,refs,selected=screen.validate_source(screen.DEFAULT_SOURCE,oldplan['source_sha256'])
    identity=next(h for h in json.loads((ROOT/'results/20261003_local_screen_571682f5/installed_weight_checksums.json').read_text(encoding='utf-8'))['models'] if h['model_key']==MODEL)
    plan={'created_utc':live.utc_now(),'classification':'OPTIONAL_PREDECLARED_DEV60_COMPATIBILITY_RETRY',
        'model':MODEL,'installed_identity':identity,'source_sha256':oldplan['source_sha256'],
        'ids':[i.item_id for i in selected],'native_load':live.safe_load.native_load_body(MODEL),
        'request':screen.payload('INSTANCE','PUBLIC_PROMPT','none'),
        'limits':{'total_wall_s':420,'load_wall_s':120,'request_wall_s':45,'ram_gib':2,'gpu_free_mib':512},
        'differences_from_historical':'Native auto placement safe checkpoint0 load; OpenAI-compatible historical DEV60 request/scorer retained. Actual GPU layers unknown.',
        'previous_failure':'20261003 manifest TimeoutExpired during20s load monitoring, guardnull, 0 modelcalls; not evidence of OOM.',
        'retry_policy':'Only one load, no repair or retry. Preserve partial/error. No full DEV293.',
        'script_sha256':live.sha(Path(__file__).read_bytes()),'paid_api_calls':0,'private_test_run':False}
    live.write_new(SAFE/'protocol_frozen.json',plan)
    (SAFE/'runner_snapshot.py.txt').write_bytes(Path(__file__).read_bytes())
    key,_=resolve_api_key('local',env_path=ROOT/'.env')
    if not key:raise RuntimeError('LocalCredentialUnavailable')
    rows=[];load={};identifier=None;started=time.perf_counter();stop=None
    async with live.httpx.AsyncClient(timeout=live.httpx.Timeout(5,read=130),trust_env=False,headers={'Authorization':'Bearer '+key}) as client:
        if any(m.get('loaded_instances') for m in await live.listing(client)):
            raise RuntimeError('OtherModelLoaded')
        try:
            before=live.snapshot();load['before']=before
            if before['system_ram_available_gib']<3 or before['gpu_memory_free_mib']<1024:
                raise RuntimeError('PreloadHeadroom')
            async def load_op():
                r=await client.post(live.BASE+'/api/v1/models/load',json=plan['native_load']);r.raise_for_status();return r.json()
            load=await live.guarded(load_op,deadline_s=120)
            if load['error']:raise RuntimeError(load['error'])
            identifier=load['value']['instance_id'];live.safe_load.verify_load_config(load['value']['load_config'])
            load['value']['load_config'].pop('prompt_template',None)
            async def infer(prompt):
                body=screen.payload(identifier,prompt,'none');began=time.perf_counter()
                async def operation():
                    lines=[]
                    async with client.stream('POST',live.BASE+'/v1/chat/completions',json=body) as response:
                        response.raise_for_status()
                        async for line in response.aiter_lines():lines.append((time.perf_counter(),line))
                    clocks=iter(t for t,line in lines if line and not line.startswith(':') and line.startswith('data:'))
                    return native.consume_sse((line for _,line in lines),began,clock=lambda:next(clocks))
                g=await live.guarded(operation,deadline_s=45);v=g.pop('value') or {}
                error=g['error'] or v.get('event_error')
                if not error and (not v.get('stream_done') or v.get('finish_reason')!='stop'):error='IncompleteCompletion'
                return {'request':body,**g,'error':{'type':error} if error else None,'prediction_raw':v.get('final_content',''),
                    'response_received':bool(v),'usage':v.get('usage',{}),'timing':v.get('timing',{}),'raw':v}
            warm=await infer(screen.WARMUP);live.append(RAW/'calls.jsonl',live.redact({'kind':'warmup',**warm},key))
            if warm['error']:raise RuntimeError('WarmupFailed')
            for item in selected:
                if time.perf_counter()-started>420:raise RuntimeError('BranchWallDeadline')
                raw=await infer(item.prompt)
                live.append(RAW/'calls.jsonl',live.redact({'item_id':item.item_id,**raw},key))
                row=screen.safe_row(item,refs[item.item_id],raw);rows.append(row)
                live.append(SAFE/'metrics.jsonl',row)
                print(json.dumps({'optional':MODEL,'id':item.item_id,'em':row['strict_set_em'],'error':raw['error']}),flush=True)
                if raw['error']:raise RuntimeError('InferenceFailureStop')
        except Exception as exc:stop={'type':type(exc).__name__,'reason':str(exc),'utc':live.utc_now()}
        finally:
            cleaned=await live.cleanup(client,MODEL,failed_load=identifier is None)
            live.write_new(SAFE/'load.json',live.redact(load,key))
            live.write_new(SAFE/'summary.json',{'model':MODEL,'status':'COMPLETE' if len(rows)==60 and not stop else 'PARTIAL_OR_BLOCKED',
                'created_utc':plan['created_utc'],'finished_utc':live.utc_now(),'elapsed_s':time.perf_counter()-started,
                'stop':stop,'cleanup':cleaned,'metrics':summarize(rows,plan['ids']),'paid_api_calls':0,'private_test_run':False})
            print(json.dumps({'optional_complete':len(rows),'stop':stop,'cleanup_success':cleaned['success']}),flush=True)


if __name__=='__main__':asyncio.run(main())
